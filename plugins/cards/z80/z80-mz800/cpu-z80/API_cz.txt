================================================================================
  cpu-z80 v0.3.1 - API dokumentace
  Presny a rychly multi-instance emulator procesoru Zilog Z-80A
================================================================================


1. PREHLED
================================================================================

Multi-instance Z80A emulator v ANSI C. Kazda CPU instance nese vlastni
callbacky a user_data - vice nezavislych instanci muze bezet soucasne.

  - Kompletni instrukcni sadu vcetne nedokumentovanych instrukci
  - Presne pocitani T-stavu (cyklu)
  - Spravne chovani vsech flagu vcetne nedokumentovanych F3/F5
  - Interni registr MEMPTR/WZ
  - Prerusovaci rezimy IM0, IM1, IM2 a NMI
  - Callback system s user_data pro pamet, I/O a preruseni
  - API kompatibilni s cpu-z80 single-v0.2 (drop-in replacement)

Soubory:
  cpu/z80.h       - Verejne API (typy, struktury, funkce)
  cpu/z80.c       - Implementace
  utils/types.h   - Typove aliasy (u8, u16, u32, s8, ...)


2. ZAKLADNI POUZITI
================================================================================

  #include "cpu/z80.h"

  static u8 ram[65536];

  static u8 mem_read(z80_t *cpu, u16 addr, int m1_state, void *data) {
      (void)cpu; (void)m1_state; (void)data;
      return ram[addr];
  }

  static void mem_write(z80_t *cpu, u16 addr, u8 value, void *data) {
      (void)cpu; (void)data;
      ram[addr] = value;
  }

  static u8 io_read(z80_t *cpu, u16 port, void *data) {
      (void)cpu; (void)data;
      return 0xFF;
  }

  static void io_write(z80_t *cpu, u16 port, u8 value, void *data) {
      (void)cpu; (void)port; (void)value; (void)data;
  }

  int main(void) {
      z80_t *cpu = z80_create(
          mem_read, NULL,
          mem_write, NULL,
          io_read, NULL,
          io_write, NULL,
          NULL, NULL       /* intread - neni potreba */
      );

      /* Nacti program do RAM... */

      /* Emuluj 1 frame (69888 T-stavu pro ZX Spectrum) */
      z80_execute(cpu, 69888);

      z80_destroy(cpu);
      return 0;
  }


3. TYPY
================================================================================

3.1 Zakladni typy (utils/types.h)
---------------------------------

  u8, u16, u32, u64   Bezznamenkove typy (uint8_t ... uint64_t)
  s8, s16, s32         Znamenkove typy (int8_t ... int32_t)

3.2 Callback typy
-----------------

  typedef u8 (*z80_mread_cb)(z80_t *cpu, u16 addr, int m1_state, void *user_data);
    Cteni bajtu z pameti. m1_state: 1 = M1 fetch (opcode), 0 = data cteni.

  typedef void (*z80_mwrite_cb)(z80_t *cpu, u16 addr, u8 value, void *user_data);
    Zapis bajtu do pameti.

  typedef u8 (*z80_pread_cb)(z80_t *cpu, u16 port, void *user_data);
    Cteni z I/O portu. port je 16bitovy (IN A,(n) -> A:n, IN r,(C) -> BC).

  typedef void (*z80_pwrite_cb)(z80_t *cpu, u16 port, u8 value, void *user_data);
    Zapis na I/O port.

  typedef u8 (*z80_intread_cb)(z80_t *cpu, void *user_data);
    Cteni vektoru preruseni z periferie (pro z80_int()).

  typedef void (*z80_intack_cb)(z80_t *cpu, void *user_data);
    INTACK signal - potvrzeni preruseni. Pro daisy chain.

  typedef void (*z80_reti_cb)(z80_t *cpu, void *user_data);
    RETI instrukce - notifikace periferii.

  typedef void (*z80_ei_cb)(z80_t *cpu, void *user_data);
    Volano pri provedeni instrukce EI.

  typedef void (*z80_di_cb)(z80_t *cpu, void *user_data);
    Volano pri provedeni instrukce DI.

  typedef void (*z80_im_cb)(z80_t *cpu, u8 new_im, void *user_data);
    Volano pri zmene IM 0/1/2. new_im je nova hodnota.

  typedef void (*z80_halt_cb)(z80_t *cpu, void *user_data);
    Volano pri prechodu CPU do HALT stavu.

  typedef void (*z80_nmi_cb)(z80_t *cpu, void *user_data);
    Volano z z80_nmi() pri assertion NMI.

  typedef enum {
      Z80_IFF_REASON_RESET    = 0,  // CPU reset
      Z80_IFF_REASON_EI       = 1,  // EI instrukce
      Z80_IFF_REASON_DI       = 2,  // DI instrukce
      Z80_IFF_REASON_INT_ACK  = 3,  // Maskovane INT prijato (jakekoliv IM)
      Z80_IFF_REASON_NMI_ACK  = 4,  // NMI prijato
      Z80_IFF_REASON_RETI     = 5,  // RETI
      Z80_IFF_REASON_RETN     = 6   // RETN
  } z80_iff_change_reason_t;
    Duvod predavany do z80_iff_change_cb.

  typedef void (*z80_iff_change_cb)(z80_t *cpu,
                                    u8 new_iff1, u8 new_iff2,
                                    u8 reason, void *user_data);
    Fire vzdy kdyz CPU modifikuje IFF1 nebo IFF2 - EI, DI, RETI, RETN,
    INT ack (jakekoliv IM), NMI ack a reset. Doplnuje ei/di/nmi/reti
    callbacky a poskytuje sjednoceny event source pro sledovani
    prerusovaciho systemu. NULL = no-op, nulovy overhead v hot path.

  typedef enum {
      Z80_CPU_CTRL_HALT_ENTER = 0,  // HALT vykonana, cpu->halted 0->1
      Z80_CPU_CTRL_HALT_EXIT  = 1,  // IRQ/NMI probudilo z HALT, cpu->halted 1->0
      Z80_CPU_CTRL_RST_00     = 2,  // 0xC7
      Z80_CPU_CTRL_RST_08     = 3,  // 0xCF
      Z80_CPU_CTRL_RST_10     = 4,  // 0xD7
      Z80_CPU_CTRL_RST_18     = 5,  // 0xDF
      Z80_CPU_CTRL_RST_20     = 6,  // 0xE7
      Z80_CPU_CTRL_RST_28     = 7,  // 0xEF
      Z80_CPU_CTRL_RST_30     = 8,  // 0xF7
      Z80_CPU_CTRL_RST_38     = 9   // 0xFF
  } z80_cpu_ctrl_event_t;
    Event ID predavany do z80_cpu_ctrl_event_cb.

  typedef void (*z80_cpu_ctrl_event_cb)(z80_t *cpu, u8 event, u16 pc,
                                        void *user_data);
    Fire pri HALT entry (op 0x76), HALT exit (IRQ/NMI ack, jen pokud
    CPU bylo skutecne v HALT) a pri dispatch kazdeho RST opcodu
    0xC7..0xFF. pc je adresa za HALT/RST opkodem (= return adresa
    pushnuta RSTem, nebo adresa kam by se CPU vratilo po probuzeni).
    Doplnuje halt_cb. NULL = no-op, nulovy overhead v hot path.

  typedef void (*z80_call_cb)(z80_t *cpu,
                              u16 call_site, u16 target, u16 return_addr,
                              void *user_data);
    Fire pri dispatch kazdeho CALL nn (0xCD) a kazdeho taken
    CALL cc, nn (0xC4..0xFC). Pri not-taken podminenem CALL nefire.
    Fire timing: tesne pred PUSH(rPC); rPC = target.
    Urceno pro callstack subsystemy (shadow stack push). NULL = no-op.

  typedef void (*z80_ret_cb)(z80_t *cpu,
                             u16 pop_target, u16 sp_before_pop,
                             void *user_data);
    Fire pri dispatch RET (0xC9) a kazdeho taken RET cc (0xC0..0xF8).
    Pri not-taken podminenem RET nefire. NEPOKRYVA RETI (ED 4D) ani
    RETN (ED 45) - ty maji reti_cb a iff_change_cb. pop_target je
    PEEK16(sp_before_pop).
    Urceno pro callstack subsystemy (shadow stack pop). NULL = no-op.

3.3 z80_reg_t
-------------

  Enum pro z80_get_reg() / z80_set_reg():
    Z80_REG_AF, Z80_REG_BC, Z80_REG_DE, Z80_REG_HL,
    Z80_REG_AF2, Z80_REG_BC2, Z80_REG_DE2, Z80_REG_HL2,
    Z80_REG_IX, Z80_REG_IY, Z80_REG_SP, Z80_REG_PC,
    Z80_REG_WZ, Z80_REG_IR

3.4 z80_t
---------

  Struktura stavu CPU. Primo pristupne registry:
    z80_pair_t af, bc, de, hl          Hlavni sada
    z80_pair_t af2, bc2, de2, hl2      Alternativni sada
    z80_pair_t ix, iy, wz              Indexove + MEMPTR
    u16 sp, pc                         Stack Pointer, Program Counter
    u8 i, r                            Interrupt Vector, Refresh
    u8 iff1, iff2, im                  Prerusovaci system
    bool halted, int_pending, nmi_pending
    u32 cycles, total_cycles           Pocitadla T-stavu
    int wait_cycles                    Extra wait states


4. FUNKCE
================================================================================

4.1 Zivotni cyklus
------------------

  z80_t *z80_create(
      z80_mread_cb mread, void *mread_data,
      z80_mwrite_cb mwrite, void *mwrite_data,
      z80_pread_cb pread, void *pread_data,
      z80_pwrite_cb pwrite, void *pwrite_data,
      z80_intread_cb intread, void *intread_data
  );
    Vytvori novou CPU instanci. Alokuje pamet, nastavi callbacky, provede reset.
    intread muze byt NULL. Vraci NULL pri chybe alokace.

  void z80_destroy(z80_t *cpu);
    Znici CPU instanci. cpu muze byt NULL (no-op).

  void z80_reset(z80_t *cpu);
    Reset CPU. Zachovava callbacky, resetuje registry a prerusovaci system.
    AF = 0xFFFF, SP = 0xFFFF, PC = 0x0000, ostatni = 0.

4.2 Emulace
-----------

  int z80_step(z80_t *cpu);
    Provede prave jednu instrukci; vrati jeji T-staty. Pouziva per-step
    jadro: hlavni registry ziji primo v cpu-> (zadny prolog/epilog
    registrove cache per volani). Pro cycle-accurate hosty synchronizujici
    periferie po jedne instrukci.

  int z80_execute(z80_t *cpu, int target_cycles);
    Provadi instrukce dokud nedosahne target_cycles T-stavu; vrati skutecny
    pocet (>= target_cycles). Pouziva batch jadro: hlavni registry jsou
    cachovane v C lokalech a prolog/epilog je amortizovan na celou davku -
    nejrychlejsi pro ne-cycle-accurate davkove provadeni, ale nevraci se do
    hostitele uprostred instrukce.

  Obe jadra se prekladaji z jednoho sdileneho zdroje (cpu/z80_execute.inc)
  a jsou bit-identicka (stejny pocet cyklu; ZEXALL prochazi pres obe). Viz
  docs/benchmark_*.txt pro rozdil propustnosti mezi nimi.

4.3 Preruseni
-------------

  void z80_int(z80_t *cpu);
    Vyvolani maskovaneho preruseni. Vektor se cte pres intread callback.

  void z80_irq(z80_t *cpu, u8 vector);
    Vyvolani maskovaneho preruseni s explicitnim vektorem.
    IM0: vector & 0x38, IM1: ignorovan, IM2: (I << 8) | (vector & 0xFE).

  void z80_nmi(z80_t *cpu);
    Vyvolani NMI (skok na 0x0066, IFF2 = IFF1, IFF1 = 0). 11T.

4.4 Pristup k registrum
------------------------

  u16 z80_get_reg(z80_t *cpu, z80_reg_t reg);
  void z80_set_reg(z80_t *cpu, z80_reg_t reg, u16 value);
  bool z80_is_halted(z80_t *cpu);

  Registry jsou take primo pristupne pres cpu->af.w, cpu->bc.h atd.

  Pozn.: dokud je CPU v HALT (z80_is_halted() == true), PC drzi adresu
  samotne instrukce HALT (realne Z80 chovani). Inkrement na navratovou
  adresu se deje az pri INT/NMI accept, takze adresa ulozena na stack
  je stale "adresa HALT + 1".

4.5 Dynamicka zmena callbacku
------------------------------

  void z80_set_mread(z80_t *cpu, z80_mread_cb fn, void *data);
  void z80_set_mwrite(z80_t *cpu, z80_mwrite_cb fn, void *data);
  void z80_set_pread(z80_t *cpu, z80_pread_cb fn, void *data);
  void z80_set_pwrite(z80_t *cpu, z80_pwrite_cb fn, void *data);
  void z80_set_intread(z80_t *cpu, z80_intread_cb fn, void *data);
  void z80_set_intack(z80_t *cpu, z80_intack_cb fn, void *data);
  void z80_set_reti(z80_t *cpu, z80_reti_cb fn, void *data);
  void z80_set_ei(z80_t *cpu, z80_ei_cb fn, void *data);
  void z80_set_di(z80_t *cpu, z80_di_cb fn, void *data);
  void z80_set_im_change(z80_t *cpu, z80_im_cb fn, void *data);
  void z80_set_halt(z80_t *cpu, z80_halt_cb fn, void *data);
  void z80_set_nmi_cb(z80_t *cpu, z80_nmi_cb fn, void *data);
  void z80_set_iff_change(z80_t *cpu, z80_iff_change_cb fn, void *data);
  void z80_set_cpu_ctrl_event(z80_t *cpu, z80_cpu_ctrl_event_cb fn, void *data);
  void z80_set_call(z80_t *cpu, z80_call_cb fn, void *data);
  void z80_set_ret(z80_t *cpu, z80_ret_cb fn, void *data);
  void z80_set_post_step(z80_t *cpu, void (*fn)(z80_t *, void *), void *data);

  Vsechny event callbacky (ei/di/im/halt/nmi/iff_change/cpu_ctrl_event/
  call/ret) maji default NULL (no-op). Pri NULL hodnote provede
  dispatch smycka jen jeden nullable branch na opkod, takze neaktivni
  callbacky maji v benchmarku zanedbatelny dopad. Callbacky preziji
  z80_reset(); pri resetu navic iff_change_cb fire s reason RESET.

4.7 Volitelny RAM fast-path (build-time, opt-in)
------------------------------------------------

  Preklad s -DCPU_Z80_RAM_FASTPATH zapne inline page-table cestu pro
  pristup do ciste RAM, ktera obchazi read/write callbacky. Bez definu
  nize uvedena funkce ani souvisejici pole z80_t neexistuji.

  void z80_set_ram_fastpath(z80_t *cpu,
                            u8 *const read_table[16],
                            u8 *const write_table[16],
                            u8 *dbus_latch, bool enabled);

    read_table / write_table: 16 polozek, 1 / 4 KiB stranka (index
      addr >> 12). Non-NULL polozka je baze 4 KiB banku ciste RAM; pristup
      jde inline pres page[addr & 0x0FFF]. NULL polozka spadne na memory
      callback (pro mapped IO / ROM / contended stranky). Tabulky se
      kopiruji do instance.
    dbus_latch: volitelny ukazatel, do nehoz fast-path read zrcadli byte
      (replikuje side efekt data-bus latche tveho read callbacku), nebo NULL.
    enabled: hlavni gate. Nastav false kdykoliv je nainstalovan logging/
      tracing callback, jehoz vedlejsi efekty fast-path neumi replikovat.

  Volej pri kazde zmene pametove mapy (banking). Vyplati se jen kdyz je
  memory callback drahy; pri trivialnim callbacku je to cista rezie
  (viz docs/benchmark_*.txt).

7. INTEGRACE DO PROJEKTU
================================================================================

  gcc -O2 -I /cesta/k/cpu-z80 -c cpu/z80.c -o z80.o
  gcc -o emulator emulator.c z80.o

  Volitelny RAM fast-path (viz 4.7):
  gcc -O2 -DCPU_Z80_RAM_FASTPATH -I /cesta/k/cpu-z80 -c cpu/z80.c -o z80.o

  Pozadavky: C99+, GCC/Clang (computed goto), little-endian platforma.
